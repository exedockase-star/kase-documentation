    <!-- Interactive Logic Script -->
    <script>
        function openUploadModal() {
            document.getElementById('uploadModal').classList.remove('hidden');
        }

        function closeUploadModal() {
            document.getElementById('uploadModal').classList.add('hidden');
        }

        // Validate file size up to 50MB
        function validateFileSize(input) {
            if (input.files && input.files[0]) {
                const fileSizeMB = input.files[0].size / (1024 * 1024);
                if (fileSizeMB > 50) {
                    alert('File size exceeds 50MB limit! Please choose a smaller file or compress the PDF/images.');
                    input.value = '';
                }
            }
        }

        function handleFormSubmit(event) {
            event.preventDefault();
            
            const title = document.getElementById('inputTitle').value;
            const category = document.getElementById('inputCategory').value;
            const date = document.getElementById('inputDate').value;
            const summary = document.getElementById('inputSummary').value;
            const fileInput = document.getElementById('inputFile');
            const fileName = fileInput.files.length > 0 ? fileInput.files[0].name : "Attached File (50MB Max)";

            let badgeBg = "bg-indigo-50 text-indigo-700";
            if(category === "Event Report") badgeBg = "bg-emerald-50 text-emerald-700";
            if(category === "Field Activity") badgeBg = "bg-amber-50 text-amber-700";

            const grid = document.getElementById('card-grid');
            const newCard = document.createElement('div');
            newCard.className = "archive-card bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden flex flex-col justify-between hover:shadow-xl transition duration-300 group";
            newCard.setAttribute('data-category', category);
            newCard.setAttribute('data-title', title.toLowerCase());

            newCard.innerHTML = `
                <div>
                    <div class="h-48 bg-slate-100 border-b border-slate-100 flex items-center justify-center text-slate-400 font-semibold text-xs relative overflow-hidden">
                        <div class="absolute inset-0 bg-gradient-to-t from-slate-900/60 to-transparent flex items-end p-4">
                            <span class="text-white text-xs font-bold backdrop-blur-md bg-white/20 px-2.5 py-1 rounded-md border border-white/10">${fileName}</span>
                        </div>
                        <i class="fa-regular fa-folder-open text-3xl text-slate-300"></i>
                    </div>
                    <div class="p-6">
                        <div class="flex items-center justify-between mb-3">
                            <span class="text-xs font-black uppercase tracking-wider ${badgeBg} px-3 py-1 rounded-full">${category}</span>
                            <span class="text-xs font-medium text-slate-400">${date}</span>
                        </div>
                        <h4 class="font-bold text-xl text-slate-900 group-hover:text-indigo-600 transition">${title}</h4>
                        <p class="text-sm text-slate-600 mt-2.5 leading-relaxed">${summary}</p>
                    </div>
                </div>
                <div class="px-6 pb-6 pt-3 border-t border-slate-100 flex items-center justify-between">
                    <span class="text-xs font-semibold text-emerald-600 flex items-center gap-1.5"><i class="fa-solid fa-circle-check"></i> Logged Successfully</span>
                    <button onclick="alert('Opening attached large file: ${fileName}')" class="inline-flex items-center gap-1.5 text-indigo-600 hover:text-indigo-800 text-sm font-bold transition">
                        <span>View Document</span> <i class="fa-solid fa-arrow-up-right-from-square text-xs"></i>
                    </button>
                </div>
            `;

            grid.prepend(newCard);
            closeUploadModal();
            document.getElementById('uploadForm').reset();
            alert('Successfully published file (up to 50MB supported) to your portal!');
        }

        function filterSelection(category) {
            const buttons = document.querySelectorAll('.tab-btn');
            buttons.forEach(btn => {
                if(btn.textContent.toLowerCase().includes(category.toLowerCase()) || (category === 'all' && btn.textContent === 'All Entries')) {
                    btn.className = "tab-btn px-4 py-2 bg-indigo-600 text-white text-sm font-semibold rounded-xl shadow-xs transition";
                } else {
                    btn.className = "tab-btn px-4 py-2 bg-slate-100 hover:bg-slate-200 text-slate-700 text-sm font-semibold rounded-xl transition";
                }
            });

            const cards = document.querySelectorAll('.archive-card');
            cards.forEach(card => {
                if (category === 'all' || card.getAttribute('data-category') === category) {
                    card.style.display = "flex";
                } else {
                    card.style.display = "none";
                }
            });
        }

        function searchCards() {
            let input = document.getElementById('searchInput').value.toLowerCase();
            let cards = document.querySelectorAll('.archive-card');
            
            cards.forEach(card => {
                let title = card.getAttribute('data-title');
                let textContent = card.innerText.toLowerCase();
                if (title.includes(input) || textContent.includes(input)) {
                    card.style.display = "flex";
                } else {
                    card.style.display = "none";
                }
            });
        }
    </script>
